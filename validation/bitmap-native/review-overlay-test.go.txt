type reviewNumber interface{~int8|~int32|~int64|~uint32|~uint64|~float32|~float64}
func reviewCmp[T reviewNumber](a,b T,op CompareOp)bool {switch op {case CompareGT:return a>b;case CompareGE:return a>=b;case CompareLT:return a<b;case CompareLE:return a<=b;case CompareEQ:return a==b};return false}
func reviewCheck[T reviewNumber](t *testing.T, edges []T, literal func([]T,memory.Bitmap,memory.Bitmap,T,CompareOp)int, column func([]T,[]T,memory.Bitmap,memory.Bitmap,memory.Bitmap,CompareOp)int){
 t.Helper(); rng:=rand.New(rand.NewPCG(19,97))
 for n:=0;n<=259;n++ {if n>130&&n<255{continue};a,b:=make([]T,n),make([]T,n);for i:=range a {a[i]=edges[rng.IntN(len(edges))];b[i]=edges[rng.IntN(len(edges))]}
 for k:=0;k<7;k++ {var av,bv memory.Bitmap;if k>0 {av=make(memory.Bitmap,memory.WordsFor(n));bv=make(memory.Bitmap,len(av));for i:=range av {switch k{case 1:av[i]=^uint64(0);bv[i]=^uint64(0);case 2:av[i]=rng.Uint64();bv[i]=rng.Uint64();case 3:av[i]=0;bv[i]=^uint64(0);case 4:av[i]=^uint64(0);bv[i]=0}};if k==5 {av=nil;for i:=range bv {bv[i]=rng.Uint64()}};if k==6 {bv=nil;for i:=range av {av[i]=rng.Uint64()}}}
 for op:=CompareGT;op<=CompareEQ;op++ {for _,threshold:=range edges {want,got:=make(memory.Bitmap,memory.WordsFor(n)+1),make(memory.Bitmap,memory.WordsFor(n)+1);want[len(want)-1]=0xcafe;for i:=range got {got[i]=^uint64(0)};got[len(got)-1]=0xcafe;c:=0;for i,x:=range a {if (av==nil||(i/64<len(av)&&av[i/64]&(1<<uint(i%64))!=0))&&reviewCmp(x,threshold,op){want[i/64]|=1<<uint(i%64);c++}};if z:=literal(a,av,got,threshold,op);z!=c||!slices.Equal(got,want){t.Fatalf("literal n=%d k=%d op=%v threshold=%v count=%d want=%d got=%x want=%x",n,k,op,threshold,z,c,got,want)}}
 if column!=nil {want,got:=make(memory.Bitmap,memory.WordsFor(n)+1),make(memory.Bitmap,memory.WordsFor(n)+1);want[len(want)-1]=0xcafe;for i:=range got {got[i]=^uint64(0)};got[len(got)-1]=0xcafe;c:=0;for i,x:=range a {if (av==nil||(i/64<len(av)&&av[i/64]&(1<<uint(i%64))!=0))&&(bv==nil||(i/64<len(bv)&&bv[i/64]&(1<<uint(i%64))!=0))&&reviewCmp(x,b[i],op){want[i/64]|=1<<uint(i%64);c++}};if z:=column(a,b,av,bv,got,op);z!=c||!slices.Equal(got,want){t.Fatalf("column n=%d k=%d op=%v count=%d want=%d got=%x want=%x",n,k,op,z,c,got,want)}}
 }}}
 // Exactly one matching lane, across every offset and the second bitmap word.
 for pos:=0;pos<129;pos++ {a,b:=make([]T,129),make([]T,129);a[pos]=1;want:=make(memory.Bitmap,3);want[pos/64]=1<<uint(pos%64);out:=make(memory.Bitmap,3);if c:=literal(a,nil,out,1,CompareEQ);c!=1||!slices.Equal(out,want){t.Fatalf("single literal bit %d got=%x",pos,out)};if column!=nil {if c:=column(a,b,nil,nil,out,CompareGT);c!=1||!slices.Equal(out,want){t.Fatalf("single column bit %d got=%x",pos,out)}}}
}

func TestReviewGuardMatrix(t *testing.T) {
 for _,mode:=range []Mode{ModeScalar,ModeAuto,ModeSIMD} {for _,w:=range []int{0,64,128,256,512,1024} {s:=newScalarSet(mode);installGeneratedArchSIMD(&s,mode,w);id,_:=ComparisonID(memory.TypeInt32,CompareGT);cap,_:=s.Capability(id);feature:=(w==128&&archsimd.X86.AVX())||(w==256&&archsimd.X86.AVX2())||(w==512&&archsimd.X86.AVX512());want:=mode!=ModeScalar&&!simd.Emulated()&&w==simd.Int64s{}.Len()*64&&feature;if cap.HardwareSIMD!=want {t.Fatalf("mode=%v width=%d cap=%+v",mode,w,cap)};if want&&cap.Backend!="archsimd-amd64-mask"{t.Fatal(cap)}}}
}
func TestReviewPortableColumnValidity(t *testing.T){
 if simd.Emulated(){t.Skip("portable native unavailable")}
 s:=newScalarSet(ModeSIMD);installInt32ColumnSIMD(&s,ModeSIMD,simd.Int64s{}.Len()*64);installUint32ColumnSIMD(&s,ModeSIMD,simd.Int64s{}.Len()*64);installFloat32ColumnSIMD(&s,ModeSIMD,simd.Int64s{}.Len()*64)
 reviewCheck(t,[]int32{-1<<31,-1,0,1,1<<31-1},s.CompareInt32,s.CompareInt32Columns)
 reviewCheck(t,[]uint32{0,1,1<<31,1<<32-1},s.CompareUint32,s.CompareUint32Columns)
 reviewCheck(t,[]float32{float32(math.NaN()),float32(math.Inf(1)),float32(math.Inf(-1)),0,-1,1,math.SmallestNonzeroFloat32},s.CompareFloat32,s.CompareFloat32Columns)
}
