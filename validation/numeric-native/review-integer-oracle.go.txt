package main
import("fmt";"math/rand";"gocolumn/kernel";"gocolumn/memory")
type number interface{~int8|~int32|~int64|~uint32|~uint64}
func reference[T number](a,b T,op kernel.CompareOp)bool{switch op{case kernel.CompareGT:return a>b;case kernel.CompareGE:return a>=b;case kernel.CompareLT:return a<b;case kernel.CompareLE:return a<=b;case kernel.CompareEQ:return a==b};panic("op")}
func check[T number](name string, edges []T, bind func(kernel.Set)func([]T,memory.Bitmap,memory.Bitmap,T,kernel.CompareOp)int){
 r:=rand.New(rand.NewSource(853));for n:=0;n<140;n++{a:=make([]T,n+2);for i:=range a{a[i]=T(r.Uint64());if i<len(edges){a[i]=edges[i]}};a=a[1:len(a)-1];valids:=[]memory.Bitmap{nil,make(memory.Bitmap,memory.WordsFor(n)),make(memory.Bitmap,memory.WordsFor(n)),make(memory.Bitmap,memory.WordsFor(n))};for i:=range valids[1]{valids[1][i]=^uint64(0);valids[2][i]=r.Uint64()};for _,v:=range valids{for _,threshold:=range edges{for op:=kernel.CompareGT;op<=kernel.CompareEQ;op++{want:=make(memory.Bitmap,memory.WordsFor(n)+2);want[len(want)-2],want[len(want)-1]=0xcafe,0xbeef;count:=0;for i,x:=range a{if(v==nil||v[i/64]&(1<<uint(i%64))!=0)&&reference(x,threshold,op){want[i/64]|=1<<uint(i%64);count++}};for _,mode:=range []kernel.Mode{kernel.ModeScalar,kernel.ModeAuto,kernel.ModeSIMD}{out:=make(memory.Bitmap,len(want));for i:=range out{out[i]=^uint64(0)};out[len(out)-2],out[len(out)-1]=0xcafe,0xbeef;got:=bind(kernel.New(mode))(a,v,out,threshold,op);if got!=count{panic(fmt.Sprintf("%s n=%d op=%s mode=%s count=%d want=%d",name,n,op,mode,got,count))};for i,x:=range out{if x!=want[i]{panic(fmt.Sprintf("%s n=%d op=%s mode=%s word=%d got=%x want=%x",name,n,op,mode,i,x,want[i]))}}}}}}};fmt.Println("PASS",name)
}
func main(){check("int8",[]int8{-128,-127,-1,0,1,126,127},func(s kernel.Set)func([]int8,memory.Bitmap,memory.Bitmap,int8,kernel.CompareOp)int{return s.CompareInt8});check("int32",[]int32{-1<<31,-1<<31+1,-1,0,1,1<<31-1},func(s kernel.Set)func([]int32,memory.Bitmap,memory.Bitmap,int32,kernel.CompareOp)int{return s.CompareInt32});check("int64",[]int64{-1<<63,-1<<63+1,-1,0,1,1<<63-1},func(s kernel.Set)func([]int64,memory.Bitmap,memory.Bitmap,int64,kernel.CompareOp)int{return s.CompareInt64});check("uint32",[]uint32{0,1,1<<31-1,1<<31,1<<31+1,1<<32-1},func(s kernel.Set)func([]uint32,memory.Bitmap,memory.Bitmap,uint32,kernel.CompareOp)int{return s.CompareUint32});check("uint64",[]uint64{0,1,1<<63-1,1<<63,1<<63+1,1<<64-1},func(s kernel.Set)func([]uint64,memory.Bitmap,memory.Bitmap,uint64,kernel.CompareOp)int{return s.CompareUint64})}
